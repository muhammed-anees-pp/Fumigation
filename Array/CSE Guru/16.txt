Q16: Rotate the array elements k times. (Right Rotation)

Ans:

#include <stdio.h>
int main(){
	int limit,num[100],rotate,temp,i,j;
	printf("Enter the limit of the array: ");
	scanf("%d",&limit);
	printf("Enter the array elements: ");
	for(i=0;i<limit;i++){
		scanf("%d",&num[i]);
	}
	printf("How many times you want to rotate: ");
	scanf("%d",&rotate);
	
	for(i=1;i<=rotate;i++){
		temp=num[limit-1];
		for(j=limit-1;j>=1;j--){
			num[j]=num[j-1];
		}
		num[0]=temp;		
	}
	
	printf("Array: ");
	for(i=0;i<limit;i++){
		printf("%d ",num[i]);
	}

	return 0;
}
