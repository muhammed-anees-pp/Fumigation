Q11: 2nd smallest element in an array

Ans:


#include <stdio.h> 
#include <limits.h>
int main(){
	int limit,i,num[100],smallN1=INT_MAX,smallN2=INT_MAX;
	printf("Enter the limit of the array: ");
	scanf("%d",&limit);
	printf("Enter the elements of array: ");
	for(i=0;i<limit;i++){
		scanf("%d",&num[i]);
	}
	
	for(i=0;i<limit;i++){
		if(num[i]<smallN1){
			smallN2=smallN1;
			smallN1=num[i];
		}

		else if(num[i]<smallN2 && num[i]>smallN1){
			smallN2=num[i];
		}
	}
	
	printf("Smallest element in the array is %d",smallN1);
	printf("\n2nd smallest element in the array is %d",smallN2);

	return 0;	
}
