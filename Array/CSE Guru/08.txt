Q8: Decimal to binary conversion

Ans:

#include <stdio.h>

void convert(int);
int main(){
	int number;
	printf("Enter a number: ");
	scanf("%d",&number);
	convert(number);

	return 0;
}

void convert(int num){
int binaryValue[50],i,remainder;

for(i=0;num!=0;i++){
remainder=num%2;
binaryValue[i]=remainder;
num=num/2;
}

for(i=i-1;i>=0;i--){
printf("%d ",binaryValue[i]);

}

}


Another way:

#include <stdio.h>
int main(){
	int i,number,binaryValue[100],remainder;
	printf("Enter a number: ");
	scanf("%d",&number);
	for(i=0;number!=0;i++){
		remainder=number%2;
		binaryValue[i]=remainder;
		number=number/2;
	}
	
	for(i=i-1;i>=0;i--){
		printf("%d ",binaryValue[i]);
	}
	return 0;
}
