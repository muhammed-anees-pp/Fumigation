Q3: Deletes an element at the given index.

Ans:

#include <stdio.h>
int main(){
	int i,limit,values[100],position;
	printf("Enter the limit of array: ");
	scanf("%d",&limit);
	printf("Enter the values of array:\n");
	for(i=0;i<limit;i++){
		scanf("%d",&values[i]);
	}
	printf("Enter the position you want to delete: ");
	scanf("%d",&position);
	for(i=position-1;i<limit;i++){
		values[i]=values[i+1];
	}
	printf("Updated array:\n");
	for(i=0;i<limit-1;i++){
		printf("%d ",values[i]);
	}
	

	return 0;
}
