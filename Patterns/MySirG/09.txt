#include <stdio.h>
int main(){
	int i,j,row=4,k;
	for(i=1;i<=row;i++){
		k=1;
		for(j=1;j<=2*row-1;j++){
			if(j<=row+1-i || j>=row-1+i){
				printf("%c",64+k);
				j<row?k++:k--;
			}
			else{
				printf(" ");
				if(j==4)
				k--;
			}
		}
		printf("\n");
	}
	return 0;
}
