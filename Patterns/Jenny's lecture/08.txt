#include <stdio.h>
int main(){
	int i,j,k,row=5;
	for(i=1;i<=row;i++){
		for(j=1;j<=row-i;j++){
			printf(" ");
		}
		for(k=1;k<=2*i-1;k++){
			printf("*");
		}
		printf("\n");
	}
	return 0;
}

i space star formula(2xi-1) 
1 4	1	2x1-1=1
2 3	3	2x2-1=3
3 2	5	2x3-1=5
4 1	7	2x4-1=7
5 0	9	2x5-1=9
